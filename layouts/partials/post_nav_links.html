{{- /* 长篇章节：同一作品内的上一章/下一章 */ -}} {{- if and (eq .Section "novels") (eq .Kind "page") -}} {{- $chapters := sort .CurrentSection.RegularPages "File.Path" -}} {{- if gt (len $chapters) 1 -}} {{- $idx := -1 -}} {{- range $i, $p := $chapters -}} {{- if eq $p.File.Path $.File.Path -}}{{- $idx = $i -}}{{- end -}} {{- end -}} {{- if ge $idx 0 -}}
{{- with (index $chapters (sub $idx 1)) -}} « 上一章 {{- .Title -}} {{- end -}} 目录 {{- $.CurrentSection.Title -}} {{- with (index $chapters (add $idx 1)) -}} 下一章 » {{- .Title -}} {{- end -}}
{{- end -}} {{- end -}} {{- else -}} {{- $pages := where site.RegularPages "Type" "in" site.Params.mainSections -}} {{- if and (gt (len $pages) 1) (in $pages . ) -}}
{{- with $pages.Next . }} « {{ i18n "prev_page" }} {{- .Name -}} {{- end }} {{- with $pages.Prev . }} {{ i18n "next_page" }} » {{- .Name -}} {{- end }}
{{- end -}} {{- end -}}
